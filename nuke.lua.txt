local _origLighting = {
    Ambient = Lighting.Ambient, OutdoorAmbient = Lighting.OutdoorAmbient,
    ColorShift_Top = Lighting.ColorShift_Top, ColorShift_Bottom = Lighting.ColorShift_Bottom,
    FogColor = Lighting.FogColor, FogStart = Lighting.FogStart, FogEnd = Lighting.FogEnd,
    Brightness = Lighting.Brightness, ClockTime = Lighting.ClockTime,
    GlobalShadows = Lighting.GlobalShadows,
}
local _lightingActive = false
local _lastLightState = nil
local function anyLightFeatureOn()
    return CFG.Fullbright or CFG.PurpleSky or CFG.Fog or CFG.UseTimeOfDay
end
local function restoreLighting()
    for k, v in pairs(_origLighting) do pcall(function() Lighting[k] = v end) end
end
local function applyLighting()
    local any = anyLightFeatureOn()
    if not any then
        if _lightingActive then restoreLighting(); _lightingActive = false end
        return
    end
    local state = tostring(CFG.Fullbright).."|"..tostring(CFG.PurpleSky).."|"..
                  tostring(CFG.Fog).."|"..tostring(CFG.UseTimeOfDay).."|"..tostring(CFG.TimeOfDay)
    if state == _lastLightState then return end
    _lastLightState = state
    if CFG.Fullbright then
        Lighting.Brightness = 3
        Lighting.Ambient = Color3.fromRGB(180,180,180)
        Lighting.OutdoorAmbient = Color3.fromRGB(180,180,180)
        Lighting.GlobalShadows = false
    else
        Lighting.Brightness = _origLighting.Brightness
        Lighting.Ambient = _origLighting.Ambient
        Lighting.OutdoorAmbient = _origLighting.OutdoorAmbient
        Lighting.GlobalShadows = _origLighting.GlobalShadows
    end
    if CFG.PurpleSky then
        Lighting.Ambient = Color3.fromRGB(100,40,160)
        Lighting.OutdoorAmbient = Color3.fromRGB(130,60,190)
        Lighting.ColorShift_Top = Color3.fromRGB(200,120,255)
        Lighting.ColorShift_Bottom = Color3.fromRGB(80,30,140)
    else
        Lighting.ColorShift_Top = _origLighting.ColorShift_Top
        Lighting.ColorShift_Bottom = _origLighting.ColorShift_Bottom
    end
    if CFG.Fog then
        Lighting.FogColor = Color3.fromRGB(120,60,180)
        Lighting.FogStart = 0; Lighting.FogEnd = 500
    else
        Lighting.FogColor = _origLighting.FogColor
        Lighting.FogStart = _origLighting.FogStart
        Lighting.FogEnd = _origLighting.FogEnd
    end
    if CFG.UseTimeOfDay then Lighting.ClockTime = CFG.TimeOfDay
    else Lighting.ClockTime = _origLighting.ClockTime end
    _lightingActive = true
end
task.spawn(function() while true do task.wait(0.5); pcall(applyLighting) end end)

local _origHandProps = {}
local ARM_NAMES = {
    "LeftHand","RightHand","Left Arm","Right Arm",
    "LeftUpperArm","RightUpperArm","LeftLowerArm","RightLowerArm",
}
local function applyHands()
    local char = LocalPlayer.Character; if not char then return end
    local want = CFG.TransparentHands or CFG.HelloKittyHands
    for _, n in ipairs(ARM_NAMES) do
        local part = char:FindFirstChild(n)
        if part and part:IsA("BasePart") then
            if _origHandProps[part] == nil then
                _origHandProps[part] = {
                    Transparency = part.Transparency,
                    LocalTransparencyModifier = part.LocalTransparencyModifier,
                    Color = part.Color,
                    Material = part.Material,
                }
            end
            local orig = _origHandProps[part]
            if want then
                part.LocalTransparencyModifier = 0.55
                part.Transparency = math.max(orig.Transparency, 0.55)
            else
                part.LocalTransparencyModifier = orig.LocalTransparencyModifier
                part.Transparency = orig.Transparency
                part.Color = orig.Color
                part.Material = orig.Material
                local d = part:FindFirstChild("_HKDecal"); if d then d:Destroy() end
                local d2 = part:FindFirstChild("_HKDecal2"); if d2 then d2:Destroy() end
            end
            if CFG.HelloKittyHands and want then
                part.Color = Color3.fromRGB(25,25,25)
                pcall(function() part.Material = Enum.Material.SmoothPlastic end)
                if not part:FindFirstChild("_HKDecal") then
                    local dec = Instance.new("Decal")
                    dec.Name = "_HKDecal"; dec.Texture = HK_TEXTURE
                    dec.Face = Enum.NormalId.Top; dec.Parent = part
                    local dec2 = Instance.new("Decal")
                    dec2.Name = "_HKDecal2"; dec2.Texture = HK_TEXTURE
                    dec2.Face = Enum.NormalId.Front; dec2.Parent = part
                end
            end
        end
    end
end
task.spawn(function() while true do task.wait(0.35); pcall(applyHands) end end)
LocalPlayer.CharacterAdded:Connect(function()
    _origHandProps = {}; task.wait(0.5); pcall(applyHands)
end)

local function hslToRgb(h,s,l)
    local function hue2rgb(p,q,t)
        if t<0 then t=t+1 end
        if t>1 then t=t-1 end
        if t<1/6 then return p+(q-p)*6*t end
        if t<1/2 then return q end
        if t<2/3 then return p+(q-p)*(2/3-t)*6 end
        return p
    end
    local r,g,b
    if s==0 then r,g,b=l,l,l
    else
        local q = l<0.5 and l*(1+s) or l+s-l*s
        local p = 2*l-q
        r = hue2rgb(p,q,h+1/3); g = hue2rgb(p,q,h); b = hue2rgb(p,q,h-1/3)
    end
    return Color3.new(r,g,b)
end

local _ammoWatched={}
local function watchAmmoValue(v)
    if _ammoWatched[v] then return end
    _ammoWatched[v]=math.max(v.Value,1)
    v.Changed:Connect(function(nv)
        if not CFG.InfAmmo then return end
        if nv<_ammoWatched[v] and nv>=0 then
            task.defer(function() pcall(function() v.Value=_ammoWatched[v] end) end)
        elseif nv>_ammoWatched[v] then _ammoWatched[v]=nv end
    end)
end
local function scanForAmmo(inst)
    for _,v in ipairs(inst:GetDescendants()) do
        if v:IsA("IntValue") or v:IsA("NumberValue") then
            local n=v.Name:lower()

               or n:find("round") or n:find("chamber") or n:find("cartridge") or n:find("shot") then
                watchAmmoValue(v)
            end
        end
    end
    inst.DescendantAdded:Connect(function(v)
        if v:IsA("IntValue") or v:IsA("NumberValue") then
            local n=v.Name:lower()
            if n:find("ammo") or n:find("bullet") or n:find("mag") or n:find("clip")
               or n:find("round") or n:find("chamber") then watchAmmoValue(v) end
        end
    end)
end

local function applyStrafeForce(hrp, hum, char, dirVec, accel)
    if dirVec.Magnitude < 0.01 then
        if strafeForce then pcall(function() strafeForce.Force = Vector3.zero end) end; return
    end
    dirVec = dirVec.Unit
    local mass=0
    for _,p in ipairs(char:GetDescendants()) do
        if p:IsA("BasePart") then mass=mass+p.AssemblyMass end
    end
    if mass<1 then mass=80 end
    local inAir = hum.FloorMaterial==Enum.Material.Air
    local effAccel = accel * (inAir and 0.5 or 1.0)
    local csv = hrp.AssemblyLinearVelocity:Dot(dirVec)
    local ve = effAccel - csv
    local force = dirVec * math.clamp(ve*mass*8, -mass*effAccel*12, mass*effAccel*12)
    pcall(function() strafeForce.Force=force end)
end

local DASH_TIME = 0.16; local DASH_GAP = 0.10
local dashBody, dashConn = nil, nil
local dashActive = false; local dashCooldown = 0
local function stopDash()
    if dashConn then dashConn:Disconnect(); dashConn = nil end
    if dashBody then pcall(function() dashBody:Destroy() end); dashBody = nil end
    dashActive = false
end
local function startDash(hrp, dirVec, speed)
    stopDash(); dashActive = true
    local bv = Instance.new("BodyVelocity")
    bv.Name = "_NukeDash"; bv.Velocity = dirVec.Unit * speed
    bv.MaxForce = Vector3.new(1e5, 0, 1e5); bv.P = 8000; bv.Parent = hrp
    dashBody = bv
    local t0 = tick()
    dashConn = RunService.Heartbeat:Connect(function()
        if not dashBody or not dashBody.Parent then stopDash(); return end
        if tick() - t0 >= DASH_TIME then stopDash(); dashCooldown = DASH_GAP end
    end)
end

RunService.Heartbeat:Connect(function(dt)
    local char=LocalPlayer.Character; if not char then stopDash(); return end
    local hrp=char:FindFirstChild("HumanoidRootPart")
    local hum=char:FindFirstChildOfClass("Humanoid")
    if not hrp or not hum or hum.Health<=0 then
        stopDash()
        if strafeForce then pcall(function() strafeForce.Force=Vector3.zero end) end; return
    end
    if not CFG.Strafe and not CFG.WASDStrafe then
        stopDash()
        if strafeForce then pcall(function() strafeForce.Force=Vector3.zero end) end; return
    end
    if dashCooldown > 0 then dashCooldown = dashCooldown - dt end
    if CFG.WASDStrafe then
        if strafeForce then pcall(function() strafeForce.Force=Vector3.zero end) end
        local look = Camera.CFrame.LookVector; local right = Camera.CFrame.RightVector
        local lookFlat = Vector3.new(look.X, 0, look.Z); local rightFlat = Vector3.new(right.X, 0, right.Z)
        if lookFlat.Magnitude > 0.01 then lookFlat = lookFlat.Unit end
        if rightFlat.Magnitude > 0.01 then rightFlat = rightFlat.Unit end
        local move = Vector3.zero
        if UserInputService:IsKeyDown(Enum.KeyCode.W) then move = move + lookFlat end
        if UserInputService:IsKeyDown(Enum.KeyCode.S) then move = move - lookFlat end
        if UserInputService:IsKeyDown(Enum.KeyCode.D) then move = move + rightFlat end
        if UserInputService:IsKeyDown(Enum.KeyCode.A) then move = move - rightFlat end
        if move.Magnitude > 0.01 and not dashActive and dashCooldown <= 0 then
            startDash(hrp, move, CFG.WASDStrafeSpeed)
        end
        return
    end
    if CFG.Strafe then
        strafeTimer=strafeTimer+dt
        if strafeTimer>=STRAFE_PERIOD then strafeTimer=0; strafeDir=-strafeDir end
        local right=Camera.CFrame.RightVector; right=Vector3.new(right.X,0,right.Z)
        if right.Magnitude>0.01 then right=right.Unit end
        applyStrafeForce(hrp, hum, char, right*strafeDir, CFG.StrafeSpeed)
    end
end)

local godBypassActive = false
local godOrigCFrame = nil
local godOrigTransparency = {}
local godOrigCanCollide = {}
local godConn = nil

local function godBypassOn()
    local char = LocalPlayer.Character; if not char then return end
    local hrp = char:FindFirstChild("HumanoidRootPart")
    local hum = char:FindFirstChildOfClass("Humanoid")
    if not hrp or not hum then return end
    godOrigCFrame = hrp.CFrame
    godOrigTransparency = {}
    godOrigCanCollide = {}
    for _, p in ipairs(char:GetDescendants()) do
        if p:IsA("BasePart") then
            godOrigTransparency[p] = p.Transparency
            godOrigCanCollide[p] = p.CanCollide
            p.CanCollide = false
            p.Transparency = 1
        end
    end
    for _, acc in ipairs(char:GetChildren()) do
        if acc:IsA("Accessory") and acc:FindFirstChild("Handle") then
            local h = acc.Handle
            godOrigTransparency[h] = h.Transparency
            godOrigCanCollide[h] = h.CanCollide
            h.Transparency = 1
            h.CanCollide = false
        end
    end
    hrp.CFrame = CFrame.new(hrp.Position.X, CFG.GodBypassY, hrp.Position.Z)
                * CFrame.Angles(math.rad(180), 0, 0)
    hrp.Anchored = true
    godConn = RunService.Heartbeat:Connect(function()
        if not CFG.GodBypass then return end
        local c = LocalPlayer.Character; if not c then return end
        local h = c:FindFirstChildOfClass("Humanoid")
        if h then h.Health = h.MaxHealth; h.PlatformStand = true end
    end)
    godBypassActive = true
end

local function godBypassOff()
    if godConn then godConn:Disconnect(); godConn = nil end
    local char = LocalPlayer.Character
    if char then
        local hrp = char:FindFirstChild("HumanoidRootPart")
        local hum = char:FindFirstChildOfClass("Humanoid")
        if hrp then
            hrp.Anchored = false
            if godOrigCFrame then hrp.CFrame = godOrigCFrame end
        end
        if hum then hum.PlatformStand = false end
        for _, p in ipairs(char:GetDescendants()) do
            if p:IsA("BasePart") then
                if godOrigTransparency[p] ~= nil then p.Transparency = godOrigTransparency[p] end
                if godOrigCanCollide[p] ~= nil then p.CanCollide = godOrigCanCollide[p] end
            end
        end
        for _, acc in ipairs(char:GetChildren()) do
            if acc:IsA("Accessory") and acc:FindFirstChild("Handle") then
                local h = acc.Handle
                if godOrigTransparency[h] ~= nil then h.Transparency = godOrigTransparency[h] end
                if godOrigCanCollide[h] ~= nil then h.CanCollide = godOrigCanCollide[h] end
            end
        end
    end
    godBypassActive = false
end

task.spawn(function()
    while true do
        task.wait(0.2)
        if CFG.GodBypass and not godBypassActive then godBypassOn()
        elseif not CFG.GodBypass and godBypassActive then godBypassOff() end
    end
end)

    if CFG.GodBypass then

    end
end)

local function spawnHitEffect(worldPos)
    if not CFG.HitEffect then return end
    local sp,vis=Camera:WorldToViewportPoint(worldPos); if not vis then return end
    local cx,cy=sp.X,sp.Y
    for i=1,6 do
        local ang=math.random()*math.pi*2; local spd=math.random(3,8)
        local vx=math.cos(ang)*spd; local vy=math.sin(ang)*spd
        local c=Drawing.new("Circle"); c.Radius=math.random(2,4); c.Filled=true; c.NumSides=6
        c.Color=Color3.fromRGB(255,math.random(50,150),0); c.Position=Vector2.new(cx,cy); c.Visible=true
        local life=0
        local conn; conn=RunService.RenderStepped:Connect(function(dt)
            life=life+dt; if life>0.25 then c:Remove(); conn:Disconnect(); return end
            c.Position=Vector2.new(cx+vx*life*20,cy+vy*life*20); c.Transparency=1-(life/0.25)
        end)
    end
end
local function hookTool(tool)
    tool.Activated:Connect(function()
        local char=LocalPlayer.Character; if not char then return end
        local origin=Camera.CFrame.Position; local dir=Camera.CFrame.LookVector*500
        local rp=RaycastParams.new(); rp.FilterDescendantsInstances={char}
        rp.FilterType=Enum.RaycastFilterType.Exclude
        local result=workspace:Raycast(origin,dir,rp)
        local endPos=result and result.Position or (origin+dir)
        if CFG.WallShoot and result then
            local rem=origin+dir-result.Position
            if rem.Magnitude>1 then
                local rp2=RaycastParams.new(); rp2.FilterDescendantsInstances={char,result.Instance}
                rp2.FilterType=Enum.RaycastFilterType.Exclude
                local r2=workspace:Raycast(result.Position+rem.Unit*0.1,rem,rp2)
                if r2 then spawnHitEffect(r2.Position); endPos=r2.Position end
            end
        end
        spawnTracer(origin,endPos)
        if result then spawnHitEffect(result.Position) end
    end)
end
LocalPlayer.CharacterAdded:Connect(function(char)
    char.ChildAdded:Connect(function(obj) if obj:IsA("Tool") then hookTool(obj) end end)
end)
if LocalPlayer.Character then
    for _,obj in ipairs(LocalPlayer.Character:GetChildren()) do

end

local AF2_Names = {}

    pcall(function() VirtualInputManager:SendKeyEvent(true, Enum.KeyCode.Tab, false, game) end)
    task.wait(0.12)
    pcall(function() VirtualInputManager:SendKeyEvent(false, Enum.KeyCode.Tab, false, game) end)

local function AF2_RefreshNames()
    if AF2_Updating then return end
    AF2_Updating = true
    AF2_PressTab()
    local names = {}
    for _, p in ipairs(Players:GetPlayers()) do
        if p ~= LocalPlayer and not isTeammate(p) then names[p.Name] = true end
    end
    AF2_Names = names
    AF2_LastUpdate = tick()
    AF2_Updating = false
end
local function AF2_GetTarget()
    if not CFG.AF2 then return nil end
    local best, bestDist = nil, math.huge
    local camPos  = Camera.CFrame.Position
    local camLook = Camera.CFrame.LookVector
    local center  = Vector2.new(Camera.ViewportSize.X/2, Camera.ViewportSize.Y/2)
    local hasNames = next(AF2_Names) ~= nil
    for _, p in ipairs(Players:GetPlayers()) do
        if p == LocalPlayer or isTeammate(p) then continue end
        if hasNames and not AF2_Names[p.Name] then continue end
        local char = p.Character; if not char then continue end
        local hum  = char:FindFirstChildOfClass("Humanoid")
        if not hum or hum.Health <= 0 then continue end
        local head = char:FindFirstChild("Head"); if not head then continue end
        local diff = head.Position - camPos
        if diff.Magnitude < 1 then continue end
        if CFG.AF2IgnoreBehind then
            local dot = camLook:Dot(diff.Unit)
            if dot < 0.05 then continue end
        end
        if CFG.AF2WallCheck and not isVisible(head.Position, char) then continue end
        local sp, on = Camera:WorldToViewportPoint(head.Position)
        if not on then continue end
        local d = (Vector2.new(sp.X, sp.Y) - center).Magnitude
        if d < bestDist then
            bestDist = d
            best = {head = head, model = char, sp = Vector2.new(sp.X, sp.Y), dist = d}
        end
    end
    return best
end
task.spawn(function()
    while true do
        task.wait(0.1)
        if CFG.AF2 then
            if tick() - AF2_LastUpdate >= CFG.AF2UpdateInterval then
                task.spawn(AF2_RefreshNames)
            end
        else AF2_Names = {} end
    end
end)

local function calcTarget(p)
    if p==LocalPlayer or isTeammate(p) then return nil end
    local char=p.Character; if not char then return nil end
    local hum=char:FindFirstChildOfClass("Humanoid")
    if not hum or hum.Health<=0 then return nil end

    local sp,on=Camera:WorldToViewportPoint(head.Position); if not on then return nil end

    local d=(Vector2.new(sp.X,sp.Y)-center).Magnitude
    if d>=CFG.FOV then return nil end
    if not isVisible(head.Position,char) then return nil end
    return {head=head,model=char,sp=Vector2.new(sp.X,sp.Y),dist=d}
end
local function refreshTarget()
    local best,bestDist=nil,math.huge
    if CFG.PriorityTarget then
        for _,p in ipairs(Players:GetPlayers()) do
            if PRIORITY[p.Name] then
                local t = calcTarget(p)
                if t and t.dist < bestDist then
                    bestDist = t.dist
                    best = {head=t.head, model=t.model, sp=t.sp, priority=true}
                end
            end
        end
        if best then cachedTarget = best; return end
    end
    for _,p in ipairs(Players:GetPlayers()) do
        local t = calcTarget(p)
        if t and t.dist < bestDist then
            bestDist = t.dist; best = {head=t.head, model=t.model, sp=t.sp}
        end
    end
    cachedTarget=best
end

local peekState = "idle"
local peekOriginCF = nil
local peekVel = nil
local peekEndTime = 0
local peekCooldown = 0
local function stopPeekVel()
    if peekVel then pcall(function() peekVel:Destroy() end); peekVel = nil end
end
local function returnToCover()
    stopPeekVel()
    local char = LocalPlayer.Character
    local hrp = char and char:FindFirstChild("HumanoidRootPart")
    if hrp and peekOriginCF then
        pcall(function() hrp.CFrame = peekOriginCF end)
    end
    peekState = "idle"
    peekCooldown = CFG.AutoPeekCooldown
end
local function startPeek()
    local char = LocalPlayer.Character; if not char then return false end
    local hrp = char:FindFirstChild("HumanoidRootPart"); if not hrp then return false end
    local target = cachedTarget
    if not target or not target.head or not target.head.Parent then return false end
    local targetPos = target.head.Position
    local myPos = hrp.Position
    local toTarget = Vector3.new(targetPos.X - myPos.X, 0, targetPos.Z - myPos.Z)
    if toTarget.Magnitude < 1 then return false end
    toTarget = toTarget.Unit
    local perp = Vector3.new(-toTarget.Z, 0, toTarget.X)
    local camRight = Camera.CFrame.RightVector
    local sign = (perp:Dot(camRight) >= 0) and 1 or -1
    peekOriginCF = hrp.CFrame
    peekVel = Instance.new("BodyVelocity")
    peekVel.MaxForce = Vector3.new(1e5, 0, 1e5)
    peekVel.P = 8000
    peekVel.Velocity = perp * sign * CFG.AutoPeekDist
    peekVel.Parent = hrp
    peekState = "peeking"
    peekEndTime = tick() + CFG.AutoPeekDuration
    return true
end
task.spawn(function()
    while true do
        task.wait(0.05)
        if not CFG.AutoPeek then
            if peekState ~= "idle" then returnToCover() end
        else
            if peekCooldown > 0 then peekCooldown = peekCooldown - 0.05 end
            if peekState == "idle" and peekCooldown <= 0
               and cachedTarget and cachedTarget.head and cachedTarget.head.Parent then
                startPeek()
            end
            if peekState == "peeking" and tick() > peekEndTime then
                local char = LocalPlayer.Character
                local tool = char and char:FindFirstChildOfClass("Tool")
                if tool then
                    pcall(function() tool:Activate() end)
                    local remote = tool:FindFirstChild("RemoteEvent") or tool:FindFirstChildOfClass("RemoteEvent")
                    if remote then pcall(function() remote:FireServer() end) end
                end
                pcall(function()
                    VirtualInputManager:SendMouseButtonEvent(
                        Camera.ViewportSize.X/2, Camera.ViewportSize.Y/2, 0, true, game, 1)
                    task.delay(0.05, function()
                        VirtualInputManager:SendMouseButtonEvent(
                            Camera.ViewportSize.X/2, Camera.ViewportSize.Y/2, 0, false, game, 1)
                    end)
                end)
                returnToCover()
            end
        end
    end
end)

local triggerCooldown=false
local function runTrigger()
    if not CFG.TriggerBot then return end
    local char=LocalPlayer.Character; if not char then return end
    local hum=char:FindFirstChildOfClass("Humanoid"); if not hum or hum.Health<=0 then return end
    local center=Vector2.new(Camera.ViewportSize.X/2,Camera.ViewportSize.Y/2)
    local hit=false
    for _,p in ipairs(Players:GetPlayers()) do
        if p==LocalPlayer or isTeammate(p) then continue end
        local c=p.Character; if not c then continue end
        local h=c:FindFirstChildOfClass("Humanoid"); if not h or h.Health<=0 then continue end
        for _,part in ipairs(c:GetChildren()) do
            if part:IsA("BasePart") then
                local sp,on=Camera:WorldToViewportPoint(part.Position)
                if on then
                    local d=(Vector2.new(sp.X,sp.Y)-center).Magnitude
                    if d<=CFG.TriggerFOV and isVisible(part.Position,c) then hit=true; break end
                end
            end
        end
        if hit then break end
    end

        triggerCooldown=true
        task.delay(CFG.TriggerDelay/1000, function()
            if CFG.TriggerBot then
                local tool=LocalPlayer.Character and LocalPlayer.Character:FindFirstChildOfClass("Tool")
                if tool then
                    local remote=tool:FindFirstChild("RemoteEvent") or tool:FindFirstChildOfClass("RemoteEvent")
                    if remote then pcall(function() remote:FireServer() end)
                    else pcall(function() tool:Activate() end) end
                end
                pcall(function()
                    VirtualInputManager:SendMouseButtonEvent(
                        Camera.ViewportSize.X/2,Camera.ViewportSize.Y/2,0,true,game,1)
                    task.delay(0.05,function()
                        VirtualInputManager:SendMouseButtonEvent(
                            Camera.ViewportSize.X/2,Camera.ViewportSize.Y/2,0,false,game,1)
                    end)
                end)
            end
            triggerCooldown=false
        end)
    end
end

local function stopWASDStrafe()
    if wasdStrafeConn then wasdStrafeConn:Disconnect(); wasdStrafeConn = nil end
    if wasdStrafeVel then pcall(function() wasdStrafeVel:Destroy() end); wasdStrafeVel = nil end
    wasdStrafeActive = false

local function startWASDStrafe()
    local char = LocalPlayer.Character; if not char then return end
    local hrp = char:FindFirstChild("HumanoidRootPart"); if not hrp then return end
    local look = Camera.CFrame.LookVector
    local right = Camera.CFrame.RightVector
    local lookFlat = Vector3.new(look.X, 0, look.Z)
    local rightFlat = Vector3.new(right.X, 0, right.Z)
    if lookFlat.Magnitude > 0.01 then lookFlat = lookFlat.Unit end
    if rightFlat.Magnitude > 0.01 then rightFlat = rightFlat.Unit end
    local move = Vector3.zero
    if UserInputService:IsKeyDown(Enum.KeyCode.W) then move = move + lookFlat end
    if UserInputService:IsKeyDown(Enum.KeyCode.S) then move = move - lookFlat end
    if UserInputService:IsKeyDown(Enum.KeyCode.D) then move = move + rightFlat end
    if UserInputService:IsKeyDown(Enum.KeyCode.A) then move = move - rightFlat end
    if move.Magnitude < 0.01 then return end
    move = move.Unit
    stopWASDStrafe()
    wasdStrafeActive = true
    wasdStrafeStart = tick()
    local lv = Instance.new("LinearVelocity")
    lv.Name = "_NukeWASDStrafe"
    lv.MaxForce = 1e6
    lv.VectorVelocity = move * CFG.WASDStrafePower
    lv.RelativeTo = Enum.ActuatorRelativeTo.World
    lv.Parent = hrp
    wasdStrafeVel = lv
    wasdStrafeConn = RunService.Heartbeat:Connect(function()
        if not CFG.WASDStrafeNew then stopWASDStrafe(); return end
        if tick() - wasdStrafeStart >= CFG.WASDStrafeDuration then stopWASDStrafe() end
    end)
end
RunService.Heartbeat:Connect(function()
    if not CFG.WASDStrafeNew then stopWASDStrafe(); return end
    if wasdStrafeActive then return end
    local hasInput = UserInputService:IsKeyDown(Enum.KeyCode.W)
                  or UserInputService:IsKeyDown(Enum.KeyCode.A)
                  or UserInputService:IsKeyDown(Enum.KeyCode.S)
                  or UserInputService:IsKeyDown(Enum.KeyCode.D)

        if UserInputService:IsKeyDown(Enum.KeyCode.Space) then vel=vel+Vector3.new(0,1,0) end
        if UserInputService:IsKeyDown(Enum.KeyCode.LeftShift) then vel=vel-Vector3.new(0,1,0) end
        if vel.Magnitude>0 then vel=vel.Unit*CFG.FlySpeed end
        flyBody.Velocity=vel
    end)
end
local function stopFly()
    if flyConn then flyConn:Disconnect(); flyConn=nil end
    if flyBody then flyBody:Destroy(); flyBody=nil end
    local char=LocalPlayer.Character
    if char then local hum=char:FindFirstChildOfClass("Humanoid"); if hum then hum.PlatformStand=false end end
end

RunService.RenderStepped:Connect(function()
    aimFrame=aimFrame+1
    local vp=Camera.ViewportSize
    local center=Vector2.new(vp.X/2,vp.Y/2)
    if CFG.CameraFOV and Camera.FieldOfView ~= CFG.CameraFOV then
        Camera.FieldOfView = CFG.CameraFOV
    end
    local rainbowHue = (tick()*0.15) % 1
    local rainbowCol = hslToRgb(rainbowHue, 1, 0.6)

    fovCircle.Visible=CFG.ShowFOV; fovCircle.Position=center; fovCircle.Radius=CFG.FOV
    fovFill.Visible=CFG.FOVFill; fovFill.Position=center; fovFill.Radius=CFG.FOV

    if CFG.CrosshairESP then
        local g=8; local l=16; local cx,cy=center.X,center.Y
        local col = CFG.RainbowCross and rainbowCol or Color3.fromRGB(255,255,255)
        local pts={{Vector2.new(cx-g-l,cy),Vector2.new(cx-g,cy)},
                   {Vector2.new(cx+g,cy),Vector2.new(cx+g+l,cy)},
                   {Vector2.new(cx,cy-g-l),Vector2.new(cx,cy-g)},
                   {Vector2.new(cx,cy+g),Vector2.new(cx,cy+g+l)}}
        for i,ln in ipairs(crossLines) do
            ln.Visible=true; ln.From=pts[i][1]; ln.To=pts[i][2]; ln.Color = col
        end
        crossDot.Visible=true; crossDot.Position=center
        crossDot.Color = CFG.RainbowCross and rainbowCol or C.acc
    else
        for _,ln in ipairs(crossLines) do ln.Visible=false end; crossDot.Visible=false
    end

    if aimFrame%3==0 then task.spawn(refreshTarget); task.spawn(runTrigger) end

    if CFG.Aim and UserInputService:IsMouseButtonPressed(Enum.UserInputType.MouseButton2) then
        if cachedTarget and cachedTarget.head and cachedTarget.head.Parent then
            pcall(function() Camera.CFrame=CFrame.new(Camera.CFrame.Position,cachedTarget.head.Position) end)
        end
    end

    if CFG.AF2 then
        local af2t = AF2_GetTarget()
        if af2t and af2t.head and af2t.head.Parent then
            pcall(function()
                Camera.CFrame = CFrame.new(Camera.CFrame.Position, af2t.head.Position)
            end)
        end
    end

    silentTarget=CFG.Silent and cachedTarget or nil

    if CFG.TargetESP and cachedTarget and cachedTarget.head and cachedTarget.head.Parent then
        local sp,on=Camera:WorldToViewportPoint(cachedTarget.head.Position)
        if on then
            local tx,ty=sp.X,sp.Y
            targetLine.Visible=true
            targetLine.From=Vector2.new(center.X,vp.Y); targetLine.To=Vector2.new(tx,ty)
            targetBox.Visible=true
            targetBox.Position=Vector2.new(tx-10,ty-10); targetBox.Size=Vector2.new(20,20)
            if cachedTarget.priority then
                targetBox.Color = Color3.fromRGB(255,220,0); targetBox.Thickness = 2
            else
                local pulse=math.abs(math.sin(tick()*4))
                targetBox.Color=Color3.fromRGB(255,math.floor(50+pulse*100),math.floor(50+pulse*50))
                targetBox.Thickness = 1
            end
        end
    else targetLine.Visible=false; targetBox.Visible=false end

    if not CFG.ESP then
        for _,d in pairs(espObjects) do
            for _,obj in ipairs(d.all) do pcall(function() obj.Visible=false end) end
            pcall(function() if d.hl then d.hl.Enabled=false end end)
        end; return
    end

    for model,d in pairs(espObjects) do
        local hum=model:FindFirstChildOfClass("Humanoid")
        local alive=hum and hum.Health>0 and model.Parent~=nil
        pcall(function()
            if d.hl then
                if d.hl.Parent~=model then d.hl.Parent=model; d.hl.Adornee=model end
                d.hl.Enabled=alive and (CFG.Chams or CFG.RainbowChams or CFG.HealthGlow)
                if CFG.RainbowChams and not d.isTeam then
                    d.hl.FillColor = rainbowCol
                elseif CFG.HealthGlow and hum then
                    local r = hum.Health / math.max(hum.MaxHealth,1)
                    if r < 0.35 then
                        d.hl.FillColor = Color3.fromRGB(255, math.floor(255*r*2), math.floor(255*r*2))
                    end
                end
            end
        end)
        if not alive then
            for _,obj in ipairs(d.all) do pcall(function() obj.Visible=false end) end; continue
        end
        local head=model:FindFirstChild("Head")
        local root=model:FindFirstChild("HumanoidRootPart") or model:FindFirstChild("Torso")
        if not head or not root then
            for _,obj in ipairs(d.all) do pcall(function() obj.Visible=false end) end; continue
        end
        local hSP,hVis=Camera:WorldToViewportPoint(head.Position+Vector3.new(0,0.7,0))
        local rSP,rVis=Camera:WorldToViewportPoint(root.Position-Vector3.new(0,3,0))
        local vis=hVis or rVis
        local camDist=(Camera.CFrame.Position-root.Position).Magnitude

        if d.nameT then d.nameT.Visible=vis and showName; d.nameT.Position=Vector2.new(hSP.X,hSP.Y-18) end
        if d.distT then d.distT.Visible=vis; d.distT.Position=Vector2.new(hSP.X,hSP.Y-6); d.distT.Text=math.floor(camDist).."m" end
        if d.headDot then
            d.headDot.Visible = vis and CFG.HeadDots
            if d.headDot.Visible then
                d.headDot.Position = Vector2.new(hSP.X, hSP.Y)
                d.headDot.Color = d.isTeam and C_GREEN or C_RED
            end
        end
        if d.dirLine then
            d.dirLine.Visible = vis and CFG.DirLines and not d.isTeam
            if d.dirLine.Visible then
                d.dirLine.From = Vector2.new(center.X, vp.Y)
                d.dirLine.To = Vector2.new(hSP.X, hSP.Y)
                d.dirLine.Color = d.isTeam and C_GREEN or C_RED
            end
        end
        if not d.isTeam and d.box then
            local w=math.clamp(1200/math.max(camDist,1),20,260)
            local top=hSP.Y; local bottom=rSP.Y
            local left=hSP.X-w/2; local right=hSP.X+w/2
            local pA=Vector2.new(left,top); local pB=Vector2.new(right,top)
            local pC=Vector2.new(right,bottom); local pD=Vector2.new(left,bottom)
            d.box.Visible=vis; d.box.PointA=pA; d.box.PointB=pB; d.box.PointC=pC; d.box.PointD=pD
            d.boxF.Visible=vis; d.boxF.PointA=pA; d.boxF.PointB=pB; d.boxF.PointC=pC; d.boxF.PointD=pD
            if CFG.DistColors then
                local t = math.clamp(camDist/300, 0, 1)
                local dcol = Color3.fromRGB(math.floor(255*(1-t)+40*t), math.floor(255*t), 60)
                d.box.Color = dcol; d.boxF.Color = dcol
            else
                d.box.Color = C_RED; d.boxF.Color = C_RED
            end
            if CFG.HealthBar and hum then
                local hp=math.max(hum.Health,0); local maxHp=math.max(hum.MaxHealth,1)
                local ratio=hp/maxHp
                local hpCol=Color3.fromRGB(math.floor(255*(1-ratio)),math.floor(255*ratio),0)
                local boxH=math.abs(bottom-top); local barX=left-5
                d.hpBg.Visible=vis; d.hpBg.From=Vector2.new(barX,top); d.hpBg.To=Vector2.new(barX,bottom)
                d.hpFill.Visible=vis; d.hpFill.Color=hpCol

                if d.hpT then d.hpT.Visible=vis; d.hpT.Color=hpCol
                    d.hpT.Text=math.floor(hp).." HP"; d.hpT.Position=Vector2.new(hSP.X,bottom+3) end
            else
                if d.hpBg then d.hpBg.Visible=false end
                if d.hpFill then d.hpFill.Visible=false end
                if d.hpT then d.hpT.Visible=false end
            end
            if CFG.Skeleton and camDist<80 then
                for i,bone in ipairs(BONES) do
                    local ln=d.bones[i]; if not ln then continue end
                    local pAp=model:FindFirstChild(bone[1]); local pBp=model:FindFirstChild(bone[2])

                    else ln.Visible=false end
                end
            else for _,ln in ipairs(d.bones) do ln.Visible=false end end
        end
    end
end)
